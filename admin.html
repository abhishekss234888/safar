const ADMIN_EMAIL = "admin@safar.com"; function isAdmin(user) { return !!user?.email && user.email.toLowerCase() === ADMIN_EMAIL.toLowerCase(); } onAuthStateChanged(auth, (user) => { if (!user) { authGateText.textContent = 'Redirecting to login…'; window.location.replace('index.html'); } else if (!isAdmin(user)) { authGateText.textContent = 'Access denied — redirecting…'; window.location.replace('dashboard.html'); } else { document.body.classList.add('authed'); } });
